1. Backend API — Express.js (Node.js)

ฟีเจอร์หลัก

จัดเก็บข้อมูลฟังก์ชัน (1000+ ตัว) ใน JSON หรือฐานข้อมูล (MongoDB)

API สำหรับเรียกดูฟังก์ชัน (list, filter)

API เรียกใช้ฟังก์ชันตาม id พร้อมรับพารามิเตอร์

ระบบ Authentication (JWT)

ระบบบันทึก Log การใช้งานและเก็บ Feedback


ตัวอย่างโครงสร้างโค้ด (ขยายจาก server.js)

const express = require('express');
const jwt = require('jsonwebtoken');
const bodyParser = require('body-parser');
const app = express();

app.use(bodyParser.json());

const SECRET_KEY = 'your_secret_key_here';

// ฟังก์ชันสมมุติ 1000 ฟังก์ชัน
const functions = require('./functions_1000.json');

// Middleware ตรวจสอบ JWT Token
function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];
  if (!token) return res.sendStatus(401);
  jwt.verify(token, SECRET_KEY, (err, user) => {
    if (err) return res.sendStatus(403);
    req.user = user;
    next();
  });
}

// สมัครและรับ Token
app.post('/api/login', (req, res) => {
  const username = req.body.username;
  const user = { name: username };
  const accessToken = jwt.sign(user, SECRET_KEY, { expiresIn: '1h' });
  res.json({ accessToken });
});

// ดูรายการฟังก์ชัน (ต้อง login)
app.get('/api/functions', authenticateToken, (req, res) => {
  res.json(functions);
});

// เรียกใช้ฟังก์ชันตาม id พร้อมพารามิเตอร์
app.post('/api/call/:id', authenticateToken, (req, res) => {
  const fn = functions.find(f => f.id === req.params.id);
  if (!fn) return res.status(404).json({ error: 'Function not found' });
  
  // ตัวอย่างการจำลองผลลัพธ์
  const params = req.body;
  res.json({ 
    result: `Executed ${fn.name} with parameters: ${JSON.stringify(params)}` 
  });
});

// เริ่มเซิร์ฟเวอร์
const PORT = process.env.PORT || 4000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});

2. Frontend — React.js ตัวอย่างเบื้องต้น

ฟีเจอร์หลัก

หน้าล็อกอิน รับ Token

หน้าแสดงรายการฟังก์ชัน (ดึงจาก API)

ฟอร์มส่งคำขอเรียกใช้ฟังก์ชัน พร้อมพารามิเตอร์

แสดงผลลัพธ์เรียลไทม์

UI สวยงาม ใช้ Tailwind CSS


ตัวอย่าง React Component (สั้น ๆ)

import React, { useState, useEffect } from 'react';
import axios from 'axios';

function App() {
  const [token, setToken] = useState('');
  const [functions, setFunctions] = useState([]);
  const [selectedFn, setSelectedFn] = useState(null);
  const [params, setParams] = useState({});
  const [result, setResult] = useState('');

  const login = async () => {
    const res = await axios.post('/api/login', { username: 'user1' });
    setToken(res.data.accessToken);
  };

  const fetchFunctions = async () => {
    const res = await axios.get('/api/functions', {
      headers: { Authorization: `Bearer ${token}` }
    });
    setFunctions(res.data);
  };

  const callFunction = async () => {
    const res = await axios.post(`/api/call/${selectedFn.id}`, params, {
      headers: { Authorization: `Bearer ${token}` }
    });
    setResult(res.data.result);
  };

  useEffect(() => {
    if (token) fetchFunctions();
  }, [token]);

  return (
    <div className="p-6 max-w-4xl mx-auto">
      {!token ? (
        <button onClick={login} className="btn-primary">Login</button>
      ) : (
        <>
          <select
            onChange={e => {
              const fn = functions.find(f => f.id === e.target.value);
              setSelectedFn(fn);
              setParams({});
              setResult('');
            }}
          >
            <option value="">Select function</option>
            {functions.map(fn => (
              <option key={fn.id} value={fn.id}>{fn.name}</option>
            ))}
          </select>

          {selectedFn && (
            <div className="mt-4">
              {selectedFn.parameters.map(param => (
                <input
                  key={param}
                  type="text"
                  placeholder={param}
                  className="border p-2 m-1"
                  onChange={e => setParams(prev => ({ ...prev, [param]: e.target.value }))}
                />
              ))}
              <button onClick={callFunction} className="btn-primary mt-2">Execute</button>
            </div>
          )}

          {result && <div className="mt-4 p-4 bg-gray-200 rounded">{result}</div>}
        </>
      )}
    </div>
  );
}

export default App;
